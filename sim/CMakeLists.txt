# THE SIMULATOR (docs/DESIGN.md): the window the runtime starts for OPENXR_SIM_DEVICE=sim. GLFW for the
# window, Clay for its layout, stb_truetype and stb_image_write for its text and its PNGs.
if(OPENXR_SIM_BUNDLED_GLFW)
    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
    add_subdirectory("${PROJECT_SOURCE_DIR}/third_party/glfw" glfw EXCLUDE_FROM_ALL)
    set(OXS_GLFW glfw)
else()
    find_package(glfw3 3.4 CONFIG QUIET)
    if(TARGET glfw)
        set(OXS_GLFW glfw)
    else()
        pkg_check_modules(GLFW3 REQUIRED IMPORTED_TARGET glfw3>=3.4)
        set(OXS_GLFW PkgConfig::GLFW3)
    endif()
endif()
find_package(OpenGL REQUIRED)
find_package(Threads REQUIRED)

add_executable(openxr-sim main.cpp app.cpp commands.cpp control.cpp link.cpp rig.cpp canvas.cpp font.cpp ui.cpp window.cpp
    clay_impl.cpp third_party_impl.c "${PROJECT_SOURCE_DIR}/common/composite.c")
if(WIN32)
    target_sources(openxr-sim PRIVATE os_win32.cpp install_win32.cpp openxr-sim.manifest)
    # THE RUNTIME INSIDE THE SIMULATOR (sim/install.cpp): openxr_sim.dll as a resource, so one .exe is
    # all there is to download, and the runtime it registers is always its own build's -- the two must
    # match (the shared memory's version check). The .rc names the DLL by path, which the resource
    # compiler opens itself: through Wine, that is Wine's name for it (Z:), which los-wine's wrappers
    # translate only in arguments, not inside a file.
    enable_language(RC)
    set(_oxs_dll "$<TARGET_FILE:openxr_sim>")
    if(NOT CMAKE_HOST_SYSTEM_NAME STREQUAL "Windows")
        set(_oxs_dll "Z:$<TARGET_FILE:openxr_sim>")
    endif()
    # One .rc per configuration where there are several (Visual Studio's generators), whose names differ.
    if(CMAKE_CONFIGURATION_TYPES)
        set(_oxs_rc "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/runtime.rc")
    else()
        set(_oxs_rc "${CMAKE_CURRENT_BINARY_DIR}/runtime.rc")
    endif()
    file(GENERATE OUTPUT "${_oxs_rc}" CONTENT
"// Generated by sim/CMakeLists.txt: the runtime this simulator installs (sim/install.cpp).\n1 RCDATA \"${_oxs_dll}\"\n")
    target_sources(openxr-sim PRIVATE "${_oxs_rc}")
    add_dependencies(openxr-sim openxr_sim)
    # Ninja knows the .rc, not the DLL it names: without this a rebuilt runtime would not reach the .exe.
    # (A property is found by the source's name, which must then hold no $<CONFIG>: one configuration; and
    # OBJECT_DEPENDS takes no generator expression, so the DLL is named where the top level puts it.)
    if(NOT CMAKE_CONFIGURATION_TYPES)
        set_source_files_properties("${_oxs_rc}" PROPERTIES OBJECT_DEPENDS "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}/openxr_sim.dll")
    endif()
    target_link_libraries(openxr-sim PRIVATE ws2_32 bcrypt shell32 ole32 gdi32 advapi32)
    # A windowed program: no console of its own when the runtime starts it.
    set_target_properties(openxr-sim PROPERTIES WIN32_EXECUTABLE ON)
    if(MSVC)
        target_link_options(openxr-sim PRIVATE /ENTRY:mainCRTStartup)
    endif()
else()
    target_sources(openxr-sim PRIVATE os_posix.cpp install_posix.cpp)
    # The manifest the window can make the user's active runtime (sim/install_posix.cpp): the installed one.
    target_compile_definitions(openxr-sim PRIVATE
        OPENXR_SIM_MANIFEST="${CMAKE_INSTALL_FULL_DATADIR}/openxr/1/openxr_sim.json")
    pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
    # The desktop's appearance from its portal (D-Bus), and whether its compositor draws the frame
    # (wayland-client, which GLFW loads anyway): sim/os.hpp.
    pkg_check_modules(DBUS REQUIRED IMPORTED_TARGET dbus-1)
    pkg_check_modules(WAYLAND_CLIENT REQUIRED IMPORTED_TARGET wayland-client)
    target_link_libraries(openxr-sim PRIVATE PkgConfig::FONTCONFIG PkgConfig::DBUS PkgConfig::WAYLAND_CLIENT)
endif()
oxs_notices_source(_oxs_notices)
target_sources(openxr-sim PRIVATE "${_oxs_notices}")
target_include_directories(openxr-sim PRIVATE . "${PROJECT_SOURCE_DIR}/common" "${PROJECT_SOURCE_DIR}/third_party/clay"
    "${PROJECT_SOURCE_DIR}/third_party/stb")
target_link_libraries(openxr-sim PRIVATE ${OXS_GLFW} OpenGL::GL Threads::Threads)
target_compile_definitions(openxr-sim PRIVATE OPENXR_SIM_VERSION="${PROJECT_VERSION}")
if(MSVC)
    target_compile_options(openxr-sim PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4 /wd4100 /utf-8>)
    # The vendored headers are not ours to warn about.
    set_source_files_properties(clay_impl.cpp third_party_impl.c PROPERTIES COMPILE_OPTIONS "/w")
else()
    target_compile_options(openxr-sim PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall -Wextra>)
    # The vendored C headers are not ours to warn about.
    set_source_files_properties(third_party_impl.c clay_impl.cpp PROPERTIES COMPILE_OPTIONS "-w")
endif()
install(TARGETS openxr-sim RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
